def get_db_connection():

# ---------------------- LOGIN PAGE -------------------------------
@app.route('/', methods=['GET','POST'])
def login():
    if request.method == 'POST':
        uid = request.form.get("userid")
        pwd = request.form.get("password")

        # MODIFY HERE → your login table name
        # Example TABLE: APP_USERS(USER_ID, PASSWORD)
        conn = get_db_connection()
        cur = conn.cursor()
        cur.execute("SELECT PASSWORD FROM APP_USERS WHERE USER_ID=?", (uid,))
        row = cur.fetchone()
        conn.close()

        if row and row[0] == pwd:
            session['user'] = uid
            return redirect(url_for("form_page"))
        else:
            return render_template("login.html", error="Invalid Username/Password")

    return render_template("login.html")

    if 'user' not in session:

@app.route('/generate_id')
def generate_id():
    try:
        conn = get_db_connection()
        cur = conn.cursor()

        # MODIFY HERE → your sequence name
        cur.execute("SELECT SEQ_RECORD_ID.NEXTVAL FROM DUAL")
        rec_id = cur.fetchone()[0]
        conn.close()

        return jsonify({'record_id': int(rec_id)})
    except Exception as e:
        return jsonify({'error': str(e)}), 500


# ---------------------- INSERT / UPDATE BUFFER -------------------
@app.route('/save_to_buffer', methods=['POST'])
def save_to_buffer():

    data = request.get_json()

    # Helper to fetch values safely
    def g(k): return data.get(k)

    # MODIFY HERE → replace with your primary field
    record_id = g("record_id")

    conn = get_db_connection()
    cur = conn.cursor()

    # MODIFY HERE → YOUR MAIN TABLE NAME
    cur.execute("SELECT 1 FROM MAIN_TABLE WHERE RECORD_ID=?", (record_id,))
    if cur.fetchone():
        return jsonify({'error': 'Record already committed!'}), 400

    # -----------------------------------------------------------------
    # MODIFY HERE → FETCH YOUR FIELDS (Example shown)
    # -----------------------------------------------------------------
    field1 = g("field1")  # e.g., customer_name
    field2 = g("field2")  # e.g., item_type
    field3 = g("field3")  # e.g., quantity
    field4 = g("field4")  # e.g., price
    field5 = g("field5")

    # Add your calculation logic here:
    total = float(field3) * float(field4)
    # -----------------------------------------------------------------

    # Check if record exists in buffer
    cur.execute("SELECT 1 FROM BUFFER_TABLE WHERE RECORD_ID=?", (record_id,))
    exists = cur.fetchone()

    if exists:
        # MODIFY HERE → UPDATE QUERY ACCORDING TO YOUR COLUMNS
        cur.execute("""
            UPDATE BUFFER_TABLE SET
                FIELD1=?, FIELD2=?, FIELD3=?, FIELD4=?, FIELD5=?, TOTAL=?
            WHERE RECORD_ID=?
        """, (field1, field2, field3, field4, field5, total, record_id))
        msg = "Record Updated Successfully"
    else:
        # MODIFY HERE → INSERT QUERY ACCORDING TO YOUR COLUMNS
        cur.execute("""
            INSERT INTO BUFFER_TABLE
            (RECORD_ID, FIELD1, FIELD2, FIELD3, FIELD4, FIELD5, TOTAL)
            VALUES (?, ?, ?, ?, ?, ?, ?)
        """, (record_id, field1, field2, field3, field4, field5, total))
        msg = "Record Inserted Successfully"

    conn.commit()
    conn.close()
    return jsonify({'message': msg})



# ---------------------- PREVIOUS RECORD --------------------------
@app.route('/previous')
def previous():
    curr = int(request.args.get("id"))

    # MODIFY HERE → your buffer table
    sql = """
        SELECT * FROM (
            SELECT * FROM BUFFER_TABLE
            WHERE RECORD_ID < ?
            ORDER BY RECORD_ID DESC
        ) WHERE ROWNUM=1
    """

    conn = get_db_connection()
    cur = conn.cursor()
    cur.execute(sql, (curr,))
    row = cur.fetchone()
    conn.close()

    if not row:
        return jsonify({'error':'No previous record'})
    
    # MODIFY HERE → return your fields
    return jsonify({'record': {
        'record_id': row[0],
        'field1': row[1],
        'field2': row[2],
        'field3': row[3],
        'field4': row[4],
        'field5': row[5],
        'total': row[6]
    }})



# ---------------------- NEXT RECORD ------------------------------
@app.route('/next')
def next():
    curr = int(request.args.get("id"))

    sql = """
        SELECT * FROM (
            SELECT * FROM BUFFER_TABLE
            WHERE RECORD_ID > ?
            ORDER BY RECORD_ID ASC
        ) WHERE ROWNUM=1
    """

    conn = get_db_connection()
    cur = conn.cursor()
    cur.execute(sql, (curr,))
    row = cur.fetchone()
    conn.close()

    if not row:
        return jsonify({'error': 'No next record'})

    return jsonify({'record': {
        'record_id': row[0],
        'field1': row[1],
        'field2': row[2],
        'field3': row[3],
        'field4': row[4],
        'field5': row[5],
        'total': row[6]
    }})



# ---------------------- DELETE -----------------------------------
@app.route('/delete', methods=['DELETE'])
def delete():
    data = request.get_json()
    rid = data.get('record_id')

    conn = get_db_connection()
    cur = conn.cursor()

    # MODIFY HERE → your tables
    cur.execute("DELETE FROM BUFFER_TABLE WHERE RECORD_ID=?", (rid,))
    cur.execute("DELETE FROM MAIN_TABLE WHERE RECORD_ID=?", (rid,))

    conn.commit()
    conn.close()

    return jsonify({'message':"Deleted"})


# ---------------------- COMMIT ALL -------------------------------
@app.route('/commit_all', methods=['POST'])
def commit_all():
    conn = get_db_connection()
    cur = conn.cursor()

    # MODIFY HERE → table names
    cur.execute("INSERT INTO MAIN_TABLE SELECT * FROM BUFFER_TABLE")
    cur.execute("DELETE FROM BUFFER_TABLE")

    conn.commit()
    conn.close()

    return jsonify({'message': 'Committed Successfully'})

// Auto ID
async function generateID() {
  let r = await fetch("/generate_id");
  let d = await r.json();
  document.getElementById('record_id').value = d.record_id;
}

// Gather data
function dataObj() {
  return {
    record_id: document.getElementById("record_id").value,
    field1: document.getElementById("field1").value,
    field2: document.getElementById("field2").value,
    field3: document.getElementById("field3").value,

  }

  let r = await fetch("/save_to_buffer", {
    method:"POST",
    headers:{'Content-Type':'application/json'},

  });

}

async function prev() {
  let id = document.getElementById("record_id").value;
  const r = await fetch(`/previous?id=${id}`);
  const d = await r.json();
  if (!d.record) return alert(d.error);

  load(d.record);
}

async function next() {
  let id = document.getElementById("record_id").value;
  const r = await fetch(`/next?id=${id}`);
  const d = await r.json();
  if (!d.record) return alert(d.error);

  load(d.record);
}

function load(r) {
  document.getElementById("record_id").value = r.record_id;

  document.getElementById("field4").value = r.field4;
  document.getElementById("field5").value = r.field5;

}
